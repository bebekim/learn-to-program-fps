cmake_minimum_required(VERSION 3.24)
project(NathansFPS VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(BUILD_GAME "Build the graphical game (requires raylib)" ON)
add_library(game_rules src/game_rules.cpp)
target_include_directories(game_rules PUBLIC include)
if(MSVC)
  target_compile_options(game_rules PRIVATE /W4)
else()
  target_compile_options(game_rules PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(rules_tests tests/rules_tests.cpp)
  target_link_libraries(rules_tests PRIVATE game_rules)
  add_test(NAME rules COMMAND rules_tests)
endif()
if(BUILD_GAME)
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  # Deliberately pinned teaching baseline; do not track a moving branch.
  FetchContent_Declare(raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG 5.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(raylib)
  add_executable(nathan_fps src/main.cpp)
  target_link_libraries(nathan_fps PRIVATE game_rules raylib)
endif()
